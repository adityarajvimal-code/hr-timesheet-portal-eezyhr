Connect Your Master Google Sheet (Free Backend)

Follow these 4 simple steps to connect your company Google Sheet so all team data saves automatically in one central spreadsheet:

  1. Open your Google Sheet and click Extensions → Apps Script in the top menu.
  2. Delete any existing code in the editor and paste the script below:
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Timesheet_Master") || ss.insertSheet("Timesheet_Master");
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Logged By", "Client", "Month", "Week", "Date", "HR Ops Tasks", "HR Ops Hours", "Founder Tasks", "Founder Hours", "Total Hours"]);
  }
  
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(),
    data.user,
    data.client,
    data.month,
    data.week,
    data.date,
    data.opsTasks,
    data.opsHours,
    data.founderTasks,
    data.founderHours,
    data.totalHours
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON);
}
  1. Click Deploy → New deployment. Select type: Web app. Set Execute as: Me, and Who has access: Anyone.
  2. Copy the generated Web App URL, log into this portal as Founder, open Master Admin → Security & Sync, and paste the URL!
